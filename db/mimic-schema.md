[schema]
-- demographic - info on patient; when counting patients, count by DEMOGRAPHIC.SUBJECT_ID
CREATE TABLE "demographic" (
"SUBJECT_ID" INTEGER,
"HADM_ID" INTEGER,
"NAME" TEXT,
"MARITAL_STATUS" TEXT,
"AGE" INTEGER,
"DOB" TEXT,
"GENDER" TEXT,
"LANGUAGE" TEXT,
"RELIGION" TEXT,
"ADMISSION_TYPE" TEXT,
"DAYS_STAY" INTEGER,
"INSURANCE" TEXT,
"ETHNICITY" TEXT,
"EXPIRE_FLAG" INTEGER,
"ADMISSION_LOCATION" TEXT,
"DISCHARGE_LOCATION" TEXT,
"DIAGNOSIS" TEXT,
"DOD" TEXT,
"DOB_YEAR" INTEGER, -- year of birth
"DOD_YEAR" INTEGER, -- year of death
"ADMITTIME" TEXT,
"DISCHTIME" TEXT,
"ADMITYEAR" INTEGER
);

-- diagnoses: diagnoses assigned to patients
CREATE TABLE "diagnoses", (
"SUBJECT_ID" INTEGER,
"HADM_ID" INTEGER,
"ICD9_CODE" TEXT,
"SHORT_TITLE" TEXT,
"LONG_TITLE" TEXT
);
-- lab: patient lab results (note HADM_ID is REAL)
CREATE TABLE "lab" (
"SUBJECT_ID" INTEGER,
"HADM_ID" INTEGER,
"ITEMID" INTEGER,
"CHARTTIME" TEXT,
"FLAG" TEXT,
"VALUE_UNIT" TEXT,
"LABEL" TEXT,
"FLUID" TEXT,
"CATEGORY" TEXT
);
-- prescriptions: medications
CREATE TABLE "prescriptions" (
"SUBJECT_ID" INTEGER,
"HADM_ID" INTEGER,
"ICUSTAY_ID" REAL,
"DRUG_TYPE" TEXT,
"DRUG" TEXT,
"FORMULARY_DRUG_CD" TEXT,
"ROUTE" TEXT,
"DRUG_DOSE" TEXT
);
-- procedures: operations performed (ICD9_CODE is INTEGER)
CREATE TABLE "procedures" (
"SUBJECT_ID" INTEGER,
"HADM_ID" INTEGER,
"ICD9_CODE" INTEGER, -- the code of the associated ICD9 diagnoses
"SHORT_TITLE" TEXT,
"LONG_TITLE" TEXT
);
