# SQL VIEWS

## DIAGNOSES

```sql

-- DIAGNOSES
SELECT ICD9_CODE, LONG_TITLE, COUNT(DISTINCT HADM_ID) n
FROM diagnoses
GROUP BY ICD9_CODE
ORDER BY n DESC
LIMIT 80;

-- LABS
SELECT LABEL, COUNT(\*) n
FROM lab
GROUP BY LABEL
ORDER BY n DESC
LIMIT 60;

-- DRUGS
SELECT UPPER(DRUG) AS DRUG, COUNT(\*) n
FROM prescriptions
GROUP BY UPPER(DRUG)
ORDER BY n DESC
LIMIT 60;

-- PROCEDURES
SELECT CAST(ICD9_CODE AS TEXT) AS ICD9_CODE, LONG_TITLE, COUNT(DISTINCT HADM_ID) n
FROM procedures
GROUP BY ICD9_CODE
ORDER BY n DESC
LIMIT 30;
```
